cmake_minimum_required(VERSION 3.25)
if(APPLE AND (NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET OR CMAKE_OSX_DEPLOYMENT_TARGET STREQUAL ""))
    set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0" CACHE STRING "Minimum macOS deployment target" FORCE)
endif()
project(CPEditorNative VERSION 0.1.0 LANGUAGES C CXX)

include(CTest)
find_package(Threads REQUIRED)
include(cmake/BoostHeaders.cmake)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(cpn_domain STATIC
    core/src/domain/text.cpp
    core/src/domain/checker.cpp
    core/src/domain/digest.cpp)
target_include_directories(cpn_domain PUBLIC core/include)
if(MSVC)
    target_compile_options(cpn_domain PUBLIC /utf-8 PRIVATE /W4)
else()
    target_compile_options(cpn_domain PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(WIN32)
    set(CPN_PROCESS_SOURCE core/src/platform/windows/process.cpp)
else()
    set(CPN_PROCESS_SOURCE core/src/platform/posix/process.cpp)
endif()
add_library(cpn_process STATIC ${CPN_PROCESS_SOURCE})
target_include_directories(cpn_process PUBLIC core/include)
target_link_libraries(cpn_process PUBLIC Threads::Threads)
if(MSVC)
    target_compile_options(cpn_process PUBLIC /utf-8 PRIVATE /W4)
else()
    target_compile_options(cpn_process PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_library(cpn_files STATIC core/src/platform/common/files.cpp)
target_include_directories(cpn_files PUBLIC core/include)
target_link_libraries(cpn_files PUBLIC cpn_domain)
if(MSVC)
    target_compile_options(cpn_files PUBLIC /utf-8)
endif()
add_library(cpn_json INTERFACE)
target_include_directories(cpn_json INTERFACE third_party/json/include)
target_link_libraries(cpn_process PUBLIC cpn_domain)
set(CPN_SERVICE_SOURCES core/src/services/profile.cpp core/src/services/batch.cpp core/src/services/json.cpp core/src/services/session.cpp core/src/services/external_checker.cpp core/src/services/companion.cpp core/src/services/companion_inbox.cpp core/src/services/companion_listener.cpp)
add_library(cpn_services STATIC ${CPN_SERVICE_SOURCES})
target_include_directories(cpn_services PUBLIC core/include)
target_link_libraries(cpn_services PUBLIC cpn_domain cpn_process cpn_files PRIVATE cpn_json cpn_transport_headers)
add_executable(cpn-cli tools/native/cpn_cli.cpp)
target_link_libraries(cpn-cli PRIVATE cpn_services cpn_json)
if(MSVC)
    target_compile_options(cpn_services PUBLIC /utf-8 PRIVATE /W4)
else()
    target_compile_options(cpn_services PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_library(cpn_jobs STATIC core/src/services/jobs.cpp)
target_link_libraries(cpn_jobs PUBLIC cpn_services)
target_include_directories(cpn_jobs PUBLIC core/include)
# 为无法解析 CMake 依赖图的原生包管理器提供单一归档。
add_library(cpn_core STATIC core/src/bridge/api.cpp core/src/services/jobs.cpp ${CPN_SERVICE_SOURCES}
    ${CPN_PROCESS_SOURCE} core/src/platform/common/files.cpp
    core/src/domain/text.cpp core/src/domain/checker.cpp core/src/domain/digest.cpp)
target_include_directories(cpn_core PUBLIC core/include PRIVATE third_party/json/include)
target_link_libraries(cpn_core PUBLIC Threads::Threads PRIVATE cpn_transport_headers)
if(MSVC)
    target_compile_options(cpn_core PUBLIC /utf-8 PRIVATE /W4)
else()
    target_compile_options(cpn_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
