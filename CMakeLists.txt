cmake_minimum_required(VERSION 3.25)
project(CPEditorNative VERSION 0.1.0 LANGUAGES CXX)

include(CTest)
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(cpn_domain STATIC
    core/src/domain/text.cpp
    core/src/domain/checker.cpp
    core/src/domain/digest.cpp)
target_include_directories(cpn_domain PUBLIC core/include)
if(MSVC)
    target_compile_options(cpn_domain PUBLIC /utf-8 PRIVATE /W4)
else()
    target_compile_options(cpn_domain PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(WIN32)
    set(CPN_PROCESS_SOURCE core/src/platform/windows/process.cpp)
else()
    set(CPN_PROCESS_SOURCE core/src/platform/posix/process.cpp)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${CPN_PROCESS_SOURCE}")
    add_library(cpn_process STATIC ${CPN_PROCESS_SOURCE})
    target_include_directories(cpn_process PUBLIC core/include)
    target_link_libraries(cpn_process PUBLIC Threads::Threads)
    if(MSVC)
        target_compile_options(cpn_process PUBLIC /utf-8 PRIVATE /W4)
    else()
        target_compile_options(cpn_process PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()

if(BUILD_TESTING)
    add_executable(cpn_domain_tests tests/core/domain_tests.cpp)
    target_link_libraries(cpn_domain_tests PRIVATE cpn_domain)
    add_test(NAME domain COMMAND cpn_domain_tests)
    if(TARGET cpn_process AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/process/process_tests.cpp")
        add_executable(cpn_process_helper tests/process/process_helper.cpp)
        add_executable(cpn_process_tests tests/process/process_tests.cpp)
        target_link_libraries(cpn_process_tests PRIVATE cpn_process)
        add_test(NAME process COMMAND cpn_process_tests $<TARGET_FILE:cpn_process_helper>)
        set_tests_properties(process PROPERTIES TIMEOUT 45)
    endif()
endif()
